    <script>
        function toggleMDU() {
            const toggle = document.getElementById('mdu-toggle');
            const container = document.getElementById('mdu-container');

            toggle.classList.toggle('active');
            container.classList.toggle('mdu-active');

            if (!toggle.classList.contains('active')) {
                const list = document.getElementById('mdu-list');
                list.querySelectorAll('.mdu-entry').forEach((entry, i) => {
                    if (i === 0) entry.querySelector('input').value = '';
                    else entry.remove();
                });
                calculateFDH();
            }
        }

        function addMDU() {
            const list = document.getElementById('mdu-list');
            const count = list.querySelectorAll('.mdu-entry').length + 1;

            const entry = document.createElement('div');
            entry.className = 'mdu-entry bg-indigo-500/5 p-5 rounded-2xl border border-indigo-500/20 relative';
            entry.innerHTML = `
                <button onclick="removeMDU(this)"
                    class="absolute top-3 right-3 text-red-400 hover:text-red-300 text-xs transition-colors"
                    title="Eliminar MDU">
                    <i class="fa-solid fa-xmark"></i>
                </button>
                <label class="block text-[10px] text-indigo-300 uppercase font-bold mb-2 tracking-widest">
                    MDU #${count} — Número de Passings
                </label>
                <input type="number" class="mdu-input tool-input border-indigo-900/50 text-white text-sm w-full"
                       placeholder="Ingresa cantidad..." oninput="calculateFDH()">
            `;
            list.appendChild(entry);
        }

        function removeMDU(btn) {
            btn.closest('.mdu-entry').remove();
            document.querySelectorAll('#mdu-list .mdu-entry').forEach((entry, i) => {
                const label = entry.querySelector('label');
                if (label) label.textContent = `MDU #${i + 1} — Número de Passings`;
            });
            calculateFDH();
        }

        function resetData() {
            document.getElementById('fdh-input').value = "";
            const mduListReset = document.getElementById('mdu-list');
            mduListReset.querySelectorAll('.mdu-entry').forEach((entry, i) => {
                if (i === 0) entry.querySelector('input').value = '';
                else entry.remove();
            });
            document.getElementById('reserve-input').value = "10";
            document.getElementById('spare-input').value = "";
            document.getElementById('splitter-type').value = "32";

            const toggle = document.getElementById('mdu-toggle');
            const container = document.getElementById('mdu-container');
            if (toggle.classList.contains('active')) {
                toggle.classList.remove('active');
                container.classList.remove('mdu-active');
            }

            calculateFDH();
        }

        function calculateFDH() {
            const ratio = parseInt(document.getElementById('splitter-type').value) || 32;
            const passings = parseFloat(document.getElementById('fdh-input').value) || 0;
            const mduPassings = Array.from(document.querySelectorAll('#mdu-list .mdu-input'))
                .reduce((sum, el) => sum + (parseFloat(el.value) || 0), 0);
            const reservePercent = parseFloat(document.getElementById('reserve-input').value) || 0;
            const spareHilos = parseFloat(document.getElementById('spare-input').value) || 0;
            
            const fdhModelElement = document.getElementById('fdh-model');
            const excessAlert = document.getElementById('excess-alert');
            const fdhDisplay = document.getElementById('fdh-display');
            const splitterElement = document.getElementById('splitter-count');
            const ratioLabel = document.getElementById('ratio-label');
            const reqVal = document.getElementById('req-val');
            const mduVal = document.getElementById('mdu-val');
            const resCalcVal = document.getElementById('res-calc-val');
            const totalVal = document.getElementById('total-val');

            ratioLabel.innerText = `(Ratio 1:${ratio})`;

            const baseTotal = passings + mduPassings;
            const reserveAmount = Math.ceil(baseTotal * (reservePercent / 100));
            const totalRequired = baseTotal + reserveAmount + spareHilos;

            // Splitters base (passings normales + reserva proporcional + spare)
            const baseReserve = Math.ceil(passings * (reservePercent / 100));
            const baseThreads = passings + baseReserve + spareHilos;
            const baseSplitters = baseThreads > 0 ? Math.ceil(baseThreads / ratio) : 0;

            // Splitters MDU: mínimo 1 por cada entrada MDU activa;
            // si los passings superan la capacidad del splitter, se calculan los necesarios
            const mduActive = document.getElementById('mdu-toggle').classList.contains('active');
            let mduSplitters = 0;
            if (mduActive) {
                document.querySelectorAll('#mdu-list .mdu-input').forEach(el => {
                    const p = parseFloat(el.value) || 0;
                    const pWithReserve = p + Math.ceil(p * (reservePercent / 100));
                    mduSplitters += Math.max(1, Math.ceil(pWithReserve / ratio));
                });
            }

            const splittersNeeded = baseSplitters + mduSplitters;
            
            reqVal.innerText = passings;
            mduVal.innerText = mduPassings;
            resCalcVal.innerText = "+" + reserveAmount;
            totalVal.innerText = Math.ceil(totalRequired);
            splitterElement.innerText = splittersNeeded;

            const models = [96, 144, 288, 432, 576, 864];
            let selectedModel = null;
            for (let model of models) {
                if (totalRequired <= model) {
                    selectedModel = model;
                    break;
                }
            }

            if (totalRequired === 0) {
                fdhModelElement.innerText = "-- CT";
                fdhModelElement.className = "text-3xl font-black text-white tracking-tighter";
                excessAlert.style.display = "none";
                fdhDisplay.style.borderColor = "#1e293b";
            } else if (selectedModel) {
                fdhModelElement.innerText = selectedModel + " CT";
                fdhModelElement.className = "text-3xl font-black text-blue-400 tracking-tighter";
                excessAlert.style.display = "none";
                fdhDisplay.style.borderColor = "#1e293b";
            } else {
                fdhModelElement.innerText = "SOBRE LÍMITE";
                fdhModelElement.className = "text-3xl font-black text-red-400 tracking-tighter";
                excessAlert.style.display = "block";
                fdhDisplay.style.borderColor = "rgba(239, 68, 68, 0.3)";
            }
        }

        function exportToExcel() {
            const ratio = parseInt(document.getElementById('splitter-type').value) || 32;
            const passings = parseFloat(document.getElementById('fdh-input').value) || 0;
            const reservePct = parseFloat(document.getElementById('reserve-input').value) || 0;
            const spare = parseFloat(document.getElementById('spare-input').value) || 0;
            const totalHilos = document.getElementById('total-val').innerText;
            const resCalc = document.getElementById('res-calc-val').innerText;
            const gabinete = document.getElementById('fdh-model').innerText;
            const splitters = document.getElementById('splitter-count').innerText;

            // Splitters base
            const baseReserve = Math.ceil(passings * (reservePct / 100));
            const baseThreads = passings + baseReserve + spare;
            const baseSplitters = baseThreads > 0 ? Math.ceil(baseThreads / ratio) : 0;

            // Datos de cada MDU
            const mduActive = document.getElementById('mdu-toggle').classList.contains('active');
            const mduEntries = mduActive
                ? Array.from(document.querySelectorAll('#mdu-list .mdu-input')).map((el, i) => {
                    const p = parseFloat(el.value) || 0;
                    const pWithReserve = p + Math.ceil(p * (reservePct / 100));
                    return { index: i + 1, passings: p, splitters: Math.max(1, Math.ceil(pWithReserve / ratio)) };
                  })
                : [];
            const mduPassingsTotal = mduEntries.reduce((s, e) => s + e.passings, 0);
            const mduSplittersTotal = mduEntries.reduce((s, e) => s + e.splitters, 0);

            // Construcción del reporte
            const sep = { "Concepto": "─────────────────────────", "Valor": "──────────" };
            const data = [
                { "Concepto": "Fecha de Reporte",             "Valor": new Date().toLocaleDateString() },
                sep,
                { "Concepto": "CONFIGURACIÓN",                "Valor": "" },
                { "Concepto": "Ratio de Splitter",            "Valor": "1:" + ratio },
                { "Concepto": "Reserva (%)",                  "Valor": reservePct + "%" },
                { "Concepto": "Spare (Hilos adicionales)",    "Valor": spare || 0 },
                sep,
                { "Concepto": "PASSINGS",                     "Valor": "" },
                { "Concepto": "Passings Base (sin MDU)",      "Valor": passings },
            ];

            if (mduActive) {
                mduEntries.forEach(e => {
                    data.push({ "Concepto": `  MDU #${e.index} — Passings`, "Valor": e.passings });
                });
                data.push({ "Concepto": "Total Passings MDU",           "Valor": mduPassingsTotal });
            }

            data.push({ "Concepto": "Reserva Calculada (Hilos)",        "Valor": resCalc });
            data.push(sep);
            data.push({ "Concepto": "RESULTADOS",                       "Valor": "" });
            data.push({ "Concepto": "TOTAL HILOS REQUERIDOS",           "Valor": totalHilos });
            data.push({ "Concepto": "GABINETE RECOMENDADO",             "Valor": gabinete });
            data.push(sep);
            data.push({ "Concepto": "SPLITTERS",                        "Valor": "" });
            data.push({ "Concepto": "Splitters Base",                   "Valor": baseSplitters });

            if (mduActive) {
                mduEntries.forEach(e => {
                    data.push({ "Concepto": `  MDU #${e.index} — Splitters (1:${ratio})`, "Valor": e.splitters });
                });
                data.push({ "Concepto": "Subtotal Splitters MDU",       "Valor": mduSplittersTotal });
            }

            data.push({ "Concepto": "TOTAL SPLITTERS",                  "Valor": parseInt(splitters) });

            const ws = XLSX.utils.json_to_sheet(data);
            const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "Resultados FDH");
            ws['!cols'] = [{ wch: 36 }, { wch: 14 }];
            XLSX.writeFile(wb, "Reporte_Calculadora_FDH.xlsx");
        }
